# 采集模块项目结构

> 范围：`collector/`。本文件展开采集业务源码；`.venv-pack`、`build`、`dist` 等依赖与构建产物只说明用途，不展开。

```text
collector
├── .claude                 # Claude 本地设置目录
├── .venv-pack              # 打包用 Python 虚拟环境，按依赖处理
├── build                   # PyInstaller 构建中间产物
├── collector               # 采集调度与任务执行核心
│   ├── __init__.py
│   ├── executor.py         # 单次采集执行器，调用设备驱动读取点位
│   ├── scheduler.py        # 采集任务调度器
│   └── task.py             # 采集任务定义与运行状态
├── config                  # 采集配置文件
│   ├── settings.fake_bootstrap.toml # 假启动/验证配置
│   ├── settings.toml.example # 配置模板(入库版本控制) → 复制为 settings.toml
│   └── settings.toml       # 本地采集配置(gitignore；连接凭据生产由 Agent env 注入)
├── core                    # 采集模块基础能力
│   ├── __init__.py
│   ├── config.py           # 配置对象与基础配置读取
│   ├── config_db.py        # 从 MySQL 加载设备、点位、分组配置
│   ├── config_loader.py    # TOML / 环境配置加载
│   ├── config_validator.py # 配置校验
│   ├── data_quality.py     # 采集数据质量与状态判断
│   ├── dry_run.py          # dry-run / 假数据启动辅助
│   ├── exceptions.py       # 采集异常类型
│   ├── logger.py           # 日志初始化
│   ├── opcda_ipc_client.py # OPC DA sidecar IPC 客户端
│   ├── opcda_ipc_protocol.py # OPC DA IPC 协议结构
│   ├── opcda_mapper.py     # OPC DA 点位映射
│   ├── retry.py            # 重试策略
│   └── system_log.py       # 系统日志上报/写入辅助
├── data_queue              # 采集内部数据队列结构
│   ├── __init__.py
│   └── types.py            # 队列消息和采集帧类型
├── device                  # 设备对象与设备管理
│   ├── __init__.py
│   ├── device.py           # 单设备抽象
│   └── manager.py          # 设备生命周期、驱动绑定与采集管理
├── dist                    # PyInstaller 输出产物
├── drivers                 # 设备协议驱动
│   ├── gesrtp-py-main      # GE SRTP 第三方/移植驱动源码与示例
│   │   ├── examples        # GE SRTP 示例脚本
│   │   ├── src             # GE SRTP 驱动实现
│   │   ├── tests           # GE SRTP 驱动测试
│   │   ├── LICENSE
│   │   ├── README.md
│   │   └── requirements.txt
│   ├── __init__.py
│   ├── base.py             # 所有驱动的统一基类
│   ├── ge_srtp_driver.py   # GE SRTP 设备驱动
│   ├── opcda_center_driver.py # OPC DA 中心化采集驱动
│   ├── opcua_driver.py     # OPC UA 驱动
│   └── s7_driver.py        # Siemens S7 驱动
├── logs                    # collector 运行日志目录
├── monitor                 # 健康检查与外部 ingest 接口
│   ├── __init__.py
│   ├── health.py           # collector 健康状态
│   └── ingest_server.py    # 外部数据写入/推送服务
├── scripts                 # 采集侧初始化与配置脚本
│   ├── agv_opcda_config.sql          # AGV OPC DA 配置 SQL
│   ├── generate_agv_opcda_config.py  # AGV OPC DA 配置生成脚本
│   └── init_opcda_center_protocol.sql # OPC DA 中心协议初始化 SQL
├── sidecars                # OPC DA Windows sidecar 程序
│   ├── opcda_proxy_service # OPC DA 代理服务，按配置连接远端 OPC DA
│   ├── opcda_push_service  # OPC DA 订阅/推送服务
│   └── opcda_sidecar       # OPC DA 探测与辅助程序
├── storage                 # Redis/存储写入层
│   ├── __init__.py
│   ├── redis_client.py     # Redis 客户端初始化
│   └── redis_writer.py     # 写入 rt:latest / rt:window / rt:stream
├── tests                   # collector 测试
│   ├── __init__.py
│   ├── conftest.py
│   ├── test_config_db_scan_rate.py
│   ├── test_config_validator.py
│   ├── test_data_quality.py
│   ├── test_device_manager.py
│   ├── test_dry_run.py
│   ├── test_ge_addressing.py
│   ├── test_integration.py
│   ├── test_interface_ingest.py
│   ├── test_opcua_driver.py
│   ├── test_redis_writer.py
│   ├── test_s7_addressing.py
│   └── test_s7_driver.py
├── check_protocols.py      # 协议配置检查脚本
├── data-watch-collector.spec # PyInstaller 打包规格
├── main.py                 # collector 启动入口
├── port_mapping.json       # 端口/设备映射配置
├── README.md               # collector 说明
├── requirements.txt        # collector Python 依赖列表
└── run.bat                 # Windows 启动脚本
```
